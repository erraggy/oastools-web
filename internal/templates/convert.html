Convert OpenAPI Specification

Convert your specification between OpenAPI versions: 2.0, 3.0, 3.1, and 3.2.

Accepts JSON or YAML files up to 2MB
Paste JSON or YAML content directly
Fetch specification from a public URL
Advanced Options
Apply overlay before conversion
Apply overlay after conversion
Usage Examples

Go Library

import "github.com/erraggy/oastools/converter"

// Simple conversion
result, err := converter.Convert("openapi.yaml", "3.1")

// With overlays
result, err := converter.ConvertWithOptions(
    converter.WithFilePath("openapi.yaml"),
    converter.WithTargetVersion("3.1"),
    converter.WithPreConversionOverlayFile("pre-overlay.yaml"),
    converter.WithPostConversionOverlayFile("post-overlay.yaml"),
)

fmt.Printf("Converted from %s to %s\n",
    result.SourceVersion, result.TargetVersion)

CLI

# Basic conversion
oastools convert openapi.yaml --target 3.1

# With overlays
oastools convert openapi.yaml --target 3.1 \
    --pre-overlay pre.yaml \
    --post-overlay post.yaml

# Output to file
oastools convert openapi.yaml --target 3.1 -o converted.yaml

Go Library

import "github.com/erraggy/oastools/differ"

// Simple diff
result, err := differ.Diff("old.yaml", "new.yaml")

// With breaking change detection
result, err := differ.DiffWithOptions(
    differ.WithFilePaths("old.yaml", "new.yaml"),
    differ.WithMode(differ.ModeBreaking),
    differ.WithIncludeInfo(true),
)

fmt.Printf("Breaking: %d, Warnings: %d\n",
    result.BreakingCount, result.WarningCount)

CLI

# Basic diff
oastools diff old.yaml new.yaml

# Breaking changes only
oastools diff --breaking old.yaml new.yaml

# With URLs
oastools diff https://api.example.com/v1/spec.yaml \
    https://api.example.com/v2/spec.yaml

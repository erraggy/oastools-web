Usage Examples

Go Library

import "github.com/erraggy/oastools/parser"

// Parse and explore the specification
result, err := parser.ParseWithOptions(
    parser.WithFilePath("openapi.yaml"),
    parser.WithResolveRefs(true),
)

// Access paths and operations
for path, item := range result.Document.Paths {
    fmt.Printf("Path: %s\n", path)
}

// Access schemas
for name, schema := range result.Document.Components.Schemas {
    fmt.Printf("Schema: %s\n", name)
}

CLI

# Explore specification structure
oastools info openapi.yaml

# List all paths
oastools info --paths openapi.yaml

# List all schemas
oastools info --schemas openapi.yaml

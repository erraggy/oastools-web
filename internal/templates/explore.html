Analyze the structure of your OpenAPI specification.

Accepts JSON or YAML files up to 2MB
Paste JSON or YAML content directly

Usage Examples

Go Library

import (
    "github.com/erraggy/oastools/parser"
    "github.com/erraggy/oastools/walker"
)

// Parse the specification
result, err := parser.ParseWithOptions(
    parser.WithFilePath("openapi.yaml"),
)

// Collect and iterate operations
ops, _ := walker.CollectOperations(result)
for _, op := range ops.All {
    fmt.Printf("%s %s\n", op.Method, op.PathTemplate)
}

// Collect and iterate schemas
schemas, _ := walker.CollectSchemas(result)
for _, s := range schemas.Components {
    fmt.Printf("Schema: %s\n", s.Name)
}

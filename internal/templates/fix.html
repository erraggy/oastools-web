Fix OpenAPI Specification

Automatically fix common issues in your OpenAPI specification.

Accepts JSON or YAML files up to 2MB
Paste JSON or YAML content directly
Fetch specification from a public URL
Advanced Options
Usage Examples

Go Library

import "github.com/erraggy/oastools/fixer"

// Apply all fixes
result, err := fixer.Fix("openapi.yaml")

// Select specific fixes
result, err := fixer.FixWithOptions(
    fixer.WithFilePath("openapi.yaml"),
    fixer.WithEnabledFixes(
        fixer.FixTypeMissingPathParameter,
        fixer.FixTypePrunedUnusedSchema,
    ),
)

fmt.Printf("Applied %d fixes\n", len(result.Fixes))

CLI

# Apply all fixes
oastools fix openapi.yaml

# Select specific fixes
oastools fix --missing-params --unused-schemas openapi.yaml

# Dry run (preview changes)
oastools fix --dry-run openapi.yaml

# Output to file
oastools fix openapi.yaml -o fixed.yaml

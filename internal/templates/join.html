Go Library

import "github.com/erraggy/oastools/joiner"

// Simple join
result, err := joiner.Join("api1.yaml", "api2.yaml")

// With options
config := joiner.DefaultConfig()
config.DefaultStrategy = joiner.StrategyRenameRight

j := joiner.New(config)
result, err := j.JoinFiles("api1.yaml", "api2.yaml", "api3.yaml")

fmt.Printf("Merged %d specs, %d collisions\n",
    len(result.Sources), result.CollisionCount)

CLI

# Basic join
oastools join api1.yaml api2.yaml

# With collision strategy
oastools join --strategy rename api1.yaml api2.yaml api3.yaml

# With overlays
oastools join api1.yaml api2.yaml \
    --pre-overlay pre.yaml \
    --post-overlay post.yaml

# Output to file
oastools join api1.yaml api2.yaml -o merged.yaml

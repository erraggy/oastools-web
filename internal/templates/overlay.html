Usage Examples

Go Library

import "github.com/erraggy/oastools/overlay"

// Apply overlay to specification
result, err := overlay.Apply("openapi.yaml", "overlay.yaml")

// From parsed documents
specDoc, _ := parser.Parse("openapi.yaml")
overlayDoc, _ := overlay.ParseOverlayFile("overlay.yaml")
result, err := overlay.ApplyParsed(specDoc, overlayDoc)

fmt.Printf("Applied %d actions\n", len(result.AppliedActions))

CLI

# Apply overlay
oastools overlay openapi.yaml overlay.yaml

# Output to file
oastools overlay openapi.yaml overlay.yaml -o modified.yaml

# Multiple overlays (applied in order)
oastools overlay openapi.yaml overlay1.yaml overlay2.yaml

Overlay Format

overlay: 1.0.0
info:
  title: My Overlay
  version: 1.0.0
actions:
  - target: $.info.title
    update:
      title: Updated API Title
  - target: $.paths['/users'].get
    update:
      summary: Get all users

Validate OpenAPI Specification

Check your specification for errors and warnings with detailed feedback.

Accepts JSON or YAML files up to 2MB
Paste JSON or YAML content directly
Fetch specification from a public URL
Advanced Options
Usage Examples

Go Library

import "github.com/erraggy/oastools/validator"

// Simple validation
result, err := validator.Validate("openapi.yaml")

// With options
result, err := validator.ValidateWithOptions(
    validator.WithFilePath("openapi.yaml"),
    validator.WithStrictMode(true),
    validator.WithIncludeWarnings(true),
)

fmt.Printf("Valid: %t, Errors: %d, Warnings: %d\n",
    result.Valid, result.ErrorCount, result.WarningCount)

CLI

# Basic validation
oastools validate openapi.yaml

# Strict mode (fail on warnings)
oastools validate --strict openapi.yaml

# Validate from URL
oastools validate https://api.example.com/openapi.yaml
